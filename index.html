function renderResultCard(payload){ const resultCard = document.getElementById("resultCard"); const leaks = payload.leaks || {}; const ranking = Object.entries(leaks) .sort((a,b)=>b[1]-a[1]) .filter(x=>x[1]>0); const rankingHTML = ranking.length ? ranking.map(([key,score]) => `
  • ${leakName(key)}: ${score}
    • `).join("") : "
  • Keine Risiken ermittelt
    • "; resultCard.classList.remove("hidden"); resultCard.innerHTML = `

    Ergebnis

    Unternehmen
    ${payload.companyName}
    Top Leak
    ${payload.topLeak || "–"}
    Benchmark
    ${payload.benchmark || "–"}
    Ranking
      ${rankingHTML}
    Nächster Schritt
    ${payload.readiness?.speed ? "Bereitschaft erfasst." : "—"}
    `; resultCard.scrollIntoView({ behavior: "smooth", block: "start" }); }
